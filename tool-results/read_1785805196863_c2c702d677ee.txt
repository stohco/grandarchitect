    46→function testEntityManager() {
    47→  console.log('Test 1: Entity Manager');
    48→  const em = createEntityManager();
    49→
    50→  // Create entities
    51→  const id1 = em.create(0n, 4);
    52→  const id2 = em.create(0n, 2);
    53→  const id3 = em.create(1n, 3);
    54→  assert(id1 !== id2, 'Two entities get different IDs');
    55→  assert(id1 > 0n, 'Entity ID is not null');
    56→  assert(em.exists(id1), 'Entity 1 exists');
    57→  assert(em.count() === 3, 'Three entities created');
    58→
    59→  // Sim-components
    60→  const transform: SimComponent = {
    61→    __type: 'transform',
    62→    __schemaHash: 'hash1',
    63→  } as any as SimComponent;
    64→  const result = em.attachSim(id1, transform);
    65→  assert(result.ok, 'Attach sim-component succeeds');
    66→  assert(em.hasSim(id1, 'transform'), 'Has transform component');
    67→  assert(!em.hasSim(id2, 'transform'), 'Entity 2 does not have transform');
    68→  const retrieved = em.getSim(id1, 'transform');
    69→  assert(retrieved?.__type === 'transform', 'Retrieved component has correct type');
    70→  assert(retrieved?.__schemaHash === 'hash1', 'Retrieved component has correct schema hash');
    71→
    72→  // Remove sim-component
    73→  const removeResult = em.removeSim(id1, 'transform');
    74→  assert(removeResult.ok, 'Remove sim-component succeeds');
    75→  assert(!em.hasSim(id1, 'transform'), 'Component removed');
    76→
    77→  // Render-components
    78→  const renderComp = { __type: 'mesh-ref' } as any;
    79→  em.attachRender(id1, renderComp);
    80→  const renderRetrieved = em.getRender(id1, 'mesh-ref');
    81→  assert(renderRetrieved?.__type === 'mesh-ref', 'Render component attached and retrieved');
    82→  em.removeRender(id1, 'mesh-ref');
    83→  assert(!em.getRender(id1, 'mesh-ref'), 'Render component removed');
    84→
    85→  // Destroy entity
    86→  const destroyResult = em.destroy(id1);
    87→  assert(destroyResult.ok, 'Destroy entity succeeds');
    88→  assert(!em.exists(id1), 'Entity 1 no longer exists');
    89→  assert(em.count() === 2, 'Two entities remain after destroy');
    90→
    91→  // Destroy non-existent
    92→  const destroyAgain = em.destroy(id1);
    93→  assert(!destroyAgain.ok, 'Destroy non-existent entity fails');
    94→
    95→  // Tier management
    96→  assert(em.getTier(id2) === 2, 'Default tier is 2');
    97→  em.setTier(id2, 4);
    98→  assert(em.getTier(id2) === 4, 'Tier updated to 4');
    99→
   100→  // Spatial node
   101→  em.setSpatialNode(id2, 42n);
   102→  assert(em.getSpatialNode(id2) === 42n, 'Spatial node updated');
   103→
   104→  // Queries
   105→  const healthComp: SimComponent = { __type: 'health', __schemaHash: 'h2' } as any;
   106→  em.attachSim(id2, healthComp);
   107→  em.attachSim(id3, healthComp);
   108→  const results = em.query({ all: ['health'] });
   109→  assert(results.length === 2, 'Query all health returns 2 entities');
   110→  const tierResults = em.query({ tierGte: 4 });
   111→  assert(tierResults.length === 1, 'Query tier >= 4 returns 1 entity');
   112→  const noneResults = em.query({ all: ['health'], none: ['transform'] });
   113→  assert(noneResults.length === 2, 'Query with none filter works');
   114→
   115→  // Query entities (full objects)
   116→  const entities = em.queryEntities({ all: ['health'] });
   117→  assert(entities.length === 2, 'queryEntities returns full entity objects');
   118→  assert(entities[0].id !== undefined, 'Entity has id');
   119→
   120→  // Compiled query
   121→  const compiled = em.compileQuery({ all: ['health'], tierGte: 2 });
   122→  assert(typeof compiled.matches === 'function', 'Compiled query has matches function');
   123→  assert(compiled.spec.all?.length === 1, 'Compiled query preserves spec');
   124→
   125→  // Clear
   126→  em.clear();
   127→  assert(em.count() === 0, 'All entities cleared');
   128→
   129→  // Entity ID allocator determinism
   130→  const alloc1 = createEntityIdAllocator();
   131→  const alloc2 = createEntityIdAllocator();
   132→  const a = alloc1.next(42n);
   133→  const b = alloc2.next(42n);
   134→  assert(a === b, 'Same seed produces same entity ID');
   135→  const c = alloc1.next(42n);
   136→  const d = alloc2.next(42n);
   137→  assert(c === d, 'Second ID also matches');
   138→  assert(a !== c, 'Sequential IDs differ');
   139→
   140→  // Null entity constant
   141→  assert(NULL_ENTITY_ID === 0n, 'NULL_ENTITY_ID is 0n');
   142→
   143→  console.log('');
   144→}

   461→function testEconomy() {
   462→  console.log('Test 4: Economy');
   463→  const api = createEconomyApi();
   464→
   465→  // Markets
   466→  const market1 = api.createMarket('village-market', 'village');
   467→  assert(market1.marketId === 'village-market', 'Market created with correct ID');
   468→  assert(api.listMarkets().length === 1, 'One market listed');
   469→  assert(api.getMarket('village-market') !== undefined, 'Market retrieved');
   470→
   471→  // Goods and prices
   472→  assert(api.setGoodPrice('village-market', 'rice', 10, 100, 80), 'Good price set');
   473→  const price = api.getGoodPrice('village-market', 'rice');
   474→  assert(price !== undefined, 'Price retrieved');
   475→  assert(price!.price === 10, 'Initial price is 10');
   476→  assert(price!.supply === 100, 'Supply is 100');
   477→  assert(price!.demand === 80, 'Demand is 80');
   478→
   479→  // Price step (Victoria II equilibrium)
   480→  api.stepPrices('village-market', 100);
