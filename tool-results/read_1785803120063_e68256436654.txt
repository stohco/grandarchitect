    43→function testEntityManager() {
    44→  console.log('Test 1: Entity Manager');
    45→  const em = createEntityManager();
    46→
    47→  // Create entities
    48→  const id1 = em.create(0n, 4);
    49→  const id2 = em.create(0n, 2);
    50→  const id3 = em.create(1n, 3);
    51→  assert(id1 !== id2, 'Two entities get different IDs');
    52→  assert(id1 > 0n, 'Entity ID is not null');
    53→  assert(em.exists(id1), 'Entity 1 exists');
    54→  assert(em.count() === 3, 'Three entities created');
    55→
    56→  // Sim-components
    57→  const transform: SimComponent = {
    58→    __type: 'transform',
    59→    __schemaHash: 'hash1',
    60→  } as any as SimComponent;
    61→  const result = em.attachSim(id1, transform);
    62→  assert(result.ok, 'Attach sim-component succeeds');
    63→  assert(em.hasSim(id1, 'transform'), 'Has transform component');
    64→  assert(!em.hasSim(id2, 'transform'), 'Entity 2 does not have transform');
    65→  const retrieved = em.getSim(id1, 'transform');
    66→  assert(retrieved?.__type === 'transform', 'Retrieved component has correct type');
    67→  assert(retrieved?.__schemaHash === 'hash1', 'Retrieved component has correct schema hash');
    68→
    69→  // Remove sim-component
    70→  const removeResult = em.removeSim(id1, 'transform');
    71→  assert(removeResult.ok, 'Remove sim-component succeeds');
    72→  assert(!em.hasSim(id1, 'transform'), 'Component removed');
    73→
    74→  // Render-components
    75→  const renderComp = { __type: 'mesh-ref' } as any;
    76→  em.attachRender(id1, renderComp);
    77→  const renderRetrieved = em.getRender(id1, 'mesh-ref');
    78→  assert(renderRetrieved?.__type === 'mesh-ref', 'Render component attached and retrieved');
    79→  em.removeRender(id1, 'mesh-ref');
    80→  assert(!em.getRender(id1, 'mesh-ref'), 'Render component removed');
    81→
    82→  // Destroy entity
    83→  const destroyResult = em.destroy(id1);
    84→  assert(destroyResult.ok, 'Destroy entity succeeds');
    85→  assert(!em.exists(id1), 'Entity 1 no longer exists');
    86→  assert(em.count() === 2, 'Two entities remain after destroy');
    87→
    88→  // Destroy non-existent
    89→  const destroyAgain = em.destroy(id1);
    90→  assert(!destroyAgain.ok, 'Destroy non-existent entity fails');
    91→
    92→  // Tier management
    93→  assert(em.getTier(id2) === 2, 'Default tier is 2');
    94→  em.setTier(id2, 4);
    95→  assert(em.getTier(id2) === 4, 'Tier updated to 4');
    96→
    97→  // Spatial node
    98→  em.setSpatialNode(id2, 42n);
    99→  assert(em.getSpatialNode(id2) === 42n, 'Spatial node updated');
   100→
   101→  // Queries
   102→  const healthComp: SimComponent = { __type: 'health', __schemaHash: 'h2' } as any;
   103→  em.attachSim(id2, healthComp);
   104→  em.attachSim(id3, healthComp);
   105→  const results = em.query({ all: ['health'] });
   106→  assert(results.length === 2, 'Query all health returns 2 entities');
   107→  const tierResults = em.query({ tierGte: 3 });
   108→  assert(tierResults.length === 1, 'Query tier >= 3 returns 1 entity');
   109→  const noneResults = em.query({ all: ['health'], none: ['transform'] });
   110→  assert(noneResults.length === 2, 'Query with none filter works');
   111→
   112→  // Query entities (full objects)
   113→  const entities = em.queryEntities({ all: ['health'] });
   114→  assert(entities.length === 2, 'queryEntities returns full entity objects');
   115→  assert(entities[0].id !== undefined, 'Entity has id');
   116→
   117→  // Compiled query
   118→  const compiled = em.compileQuery({ all: ['health'], tierGte: 2 });
   119→  assert(typeof compiled.matches === 'function', 'Compiled query has matches function');
   120→  assert(compiled.spec.all?.length === 1, 'Compiled query preserves spec');
   121→
   122→  // Clear
   123→  em.clear();
   124→  assert(em.count() === 0, 'All entities cleared');
   125→
   126→  // Entity ID allocator determinism
   127→  const alloc1 = createEntityIdAllocator();
   128→  const alloc2 = createEntityIdAllocator();
   129→  const a = alloc1.next(42n);
   130→  const b = alloc2.next(42n);
   131→  assert(a === b, 'Same seed produces same entity ID');
   132→  const c = alloc1.next(42n);
   133→  const d = alloc2.next(42n);
   134→  assert(c === d, 'Second ID also matches');
   135→  assert(a !== c, 'Sequential IDs differ');
   136→
   137→  // Null entity constant
   138→  assert(NULL_ENTITY_ID === 0n, 'NULL_ENTITY_ID is 0n');
   139→
   140→  console.log('');
   141→}

   455→function testEconomy() {
   456→  console.log('Test 4: Economy');
   457→  const api = createEconomyApi();
   458→
   459→  // Markets
   460→  const market1 = api.createMarket('village-market', 'village');
   461→  assert(market1.marketId === 'village-market', 'Market created with correct ID');
   462→  assert(api.listMarkets().length === 1, 'One market listed');
   463→  assert(api.getMarket('village-market') !== undefined, 'Market retrieved');
   464→
   465→  // Goods and prices
   466→  assert(api.setGoodPrice('village-market', 'rice', 10, 100, 80), 'Good price set');
   467→  const price = api.getGoodPrice('village-market', 'rice');
   468→  assert(price !== undefined, 'Price retrieved');
   469→  assert(price!.price === 10, 'Initial price is 10');
   470→  assert(price!.supply === 100, 'Supply is 100');
   471→  assert(price!.demand === 80, 'Demand is 80');
   472→
   473→  // Price step (Victoria II equilibrium)
   474→  api.stepPrices('village-market', 100);
   475→  const afterStep = api.getGoodPrice('village-market', 'rice');
   476→  // demand < supply, so price should decrease
   477→  assert(afterStep!.price < 10, 'Price decreased when supply > demand');
   478→
   479→  // Demand > supply → price increases
   480→  api.setGoodPrice('village-market', 'iron', 50, 30, 100);
   481→ api.stepPrices('village-market', 101);
   482→  const ironPrice = api.getGoodPrice('village-market', 'iron');
   483→  assert(ironPrice!.price > 50, 'Price increased when demand > supply');
