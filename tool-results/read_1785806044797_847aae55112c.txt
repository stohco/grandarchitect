    83→function assert(condition: boolean, label: string) {
    84→  if (condition) {
    85→    passed++;
    86→    console.log(`  \u2705 ${label}`);
    87→  } else {
    88→    failed++;
    89→    console.error(`  \u274c ${label}`);
    90→  }
    91→}
    92→
    93→function assertEq<T>(actual: T, expected: T, label: string) {
    94→  if (JSON.stringify(actual) === JSON.stringify(expected)) {
    95→    passed++;
    96→    console.log(`  \u2705 ${label}`);
    97→  } else {
    98→    failed++;
    99→    console.error(`  \u274c ${label} — expected ${JSON.stringify(expected)}, got ${JSON.stringify(actual)}`);
   100→  }
   101→}
   102→
   103→function approxEq(a: number, b: number, eps: number, label: string) {
   104→  const diff = a - b;
   105→  const abs = diff >= 0 ? diff : -diff;
   106→  if (abs <= eps) {
   107→    passed++;
   108→    console.log(`  \u2705 ${label}`);
   109→  } else {
   110→    failed++;
   111→    console.error(`  \u274c ${label} — expected ~${b}, got ${a}, diff=${abs}`);
   112→  }
   113→}
