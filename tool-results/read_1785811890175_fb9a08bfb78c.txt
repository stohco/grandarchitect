   693→export function evaluateCondition(cond: DialogueCondition, ws: WorldStateSnapshot): boolean {
   694→
   695→  switch (cond.type) {
   696→
   697→    case 'perception_state':
   698→
   699→      return ws.perceptionState === (cond.state as PerceptionState);
   700→
   701→    case 'knowledge_fact':
   702→
   703→      return cond.factId ? ws.knowledgeFacts.includes(cond.factId) : false;
   704→
   705→    case 'relationship':
   706→
   707→      if (!cond.target) return false;
   708→
   709→      const rel = ws.relationships[keyOf(cond.target)];
   710→
   711→      return rel !== undefined && rel >= (cond.min ?? 0);
   712→
   713→    case 'realm': {
   714→
   715→      const realmOrder: Realm[] = ['mortal', 'qi_induction', 'qi_condensation', 'foundation_establishment', 'core_formation', 'nascent_soul', 'spirit_severance', 'void_amalgamation', 'tribulation_crossing', 'mahayana'];
   716→
   717→      const playerIdx = realmOrder.indexOf(ws.playerRealm);
   718→
   719→      const minIdx = realmOrder.indexOf((cond.state ?? 'mortal') as Realm);
   720→
   721→      return playerIdx >= minIdx;
   722→
   723→    }
   724→
   725→    case 'quest_state':
   726→
   727→      return true; // Simplified — quests are managed by the API
   728→
   729→    case 'flag':
   730→
   731→      return cond.flag ? (ws.flags[cond.flag] ?? false) === (cond.value ?? true) : false;
   732→
   733→    case 'inventory':
   734→
   735→      return cond.itemId ? (ws.inventory[cond.itemId] ?? 0) >= (cond.min ?? 1) : false;
   736→
   737→    default:
   738→
   739→      return true;
   740→
   741→  }
   742→
   743→}

   746→
   747→export function getAvailableResponses(tree: DialogueTree, nodeId: string, ws: WorldStateSnapshot): DialogueResponse[] {
   748→
   749→  const node = tree.nodes[nodeId];
   750→
   751→  if (!node) return [];
   752→
   753→  const available: DialogueResponse[] = [];
   754→
   755→  for (const resp of node.responses) {
   756→
   757→    const allMet = resp.responseConditions.every(c => evaluateCondition(c, ws));
   758→
   759→    if (allMet) available.push(resp);
   760→
   761→  }
   762→
   763→  // Always ensure at least a "Leave" option (per doc 34 §1.2)
   764→
   765→  if (available.length === 0 && nodeId !== 'END') {
   766→
   767→    available.push({
   768→
   769→      responseId: 'leave',
   770→
   771→      playerLine: '(Leave)',
   772→
   773→      responseConditions: [],
   774→
   775→      consequences: [],
   776→
   777→      nextNodeId: 'END',
   778→
   779→    });
   780→
   781→  }
   782→
   783→  return available;
   784→
   785→}

   805→export function computeSmoothedApproval(approval: number, history: SharedExperience[], windowDays: number): number {
   806→
   807→  if (history.length === 0) return approval;
   808→
   809→  const recentDelta = history
   810→
   811→    .slice(-10) // simplified: last 10 experiences
   812→
   813→    .reduce((sum, e) => sum + e.approvalDelta, 0);
   814→
   815→  const weight = Math.min(1, history.length / 10);
   816→
   817→  return approval * (1 - weight * 0.3) + (approval + recentDelta / 10) * (weight * 0.3);
   818→
   819→}

   823→export function canExchangeCut(romance: RomanceState, currentTick: number, maxPerDay: number): boolean {
   824→
   825→  if (romance.stage === 'ended') return false;
   826→
   827→  if (romance.lastCutTick !== null && (currentTick - romance.lastCutTick) < 24 * 60) return false;
   828→
   829→  return true;
   830→
   831→}

   835→export function applyCut(romance: RomanceState, cut: Omit<CutExchange, 'cutId'>): RomanceState {
   836→
   837→  return {
   838→
   839→    ...romance,
   840→
   841→    cutsExchanged: [...romance.cutsExchanged, { ...cut, cutId: `cut-${romance.cutsExchanged.length}` }],
   842→
   843→    intensity: clamp(romance.intensity + cut.intensityDelta, 0, 1),
   844→
   845→    stability: clamp(romance.stability + cut.stabilityDelta, 0, 1),
   846→
   847→    honesty: clamp(romance.honesty + cut.honestyDelta, 0, 1),
   848→
   849→    lastCutTick: cut.tick,
   850→
   851→  };
   852→
   853→}

  1267→    if (newStage === 'acquaintance' && updated.intensity > 0.3) newStage = 'tension';
  1268→    if (newStage === 'tension' && updated.cutsExchanged.length >= 1) newStage = 'first_cut';
  1269→    if (newStage === 'first_cut' && updated.intensity > 0.6 && updated.honesty > 0.5) newStage = 'courtship';
  1270→    if (newStage === 'courtship' && updated.stability > 0.7 && updated.honesty > 0.7) newStage = 'commitment';
  1271→    if (newStage === 'commitment' && updated.stability > 0.9) newStage = 'marriage';
  1272→    if (newStage !== 'ended' && updated.stability < 0.2 && updated.intensity < 0.2) newStage = 'estrangement';
  1273→    if (updated.stability < 0.05) newStage = 'ended';
  1274→    updated.stage = newStage;
  1275→    romances.set(romanceId, updated);
  1276→    return true;

  1331→  function checkBeats(ws: WorldStateSnapshot, currentTick: number): string[] {
  1332→
  1333→    if (!spine) return [];
  1334→
  1335→    const fired: string[] = [];
  1336→
  1337→    for (const act of spine.acts) {
  1338→
  1339→    if (act.actNumber !== spine.currentAct) continue;
  1340→
  1341→    for (const beat of act.beats) {
  1342→
  1343→      if (beat.fired) continue;
  1344→
  1345→      // Simplified: check session range
  1346→
  1347→      const session = Math.floor(currentTick / (60 * 60 * 24)); // rough session count
  1348→
  1349→      if (session >= beat.sessionRange[1] && beat.missable) {
  1350→
  1351→        // Missed: session exceeded max
  1352→
  1353→        spine.beatsMissed.push(beat.beatId);
  1354→
  1355→        beat.fired = true;
  1356→
  1357→        beat.firedAtTick = currentTick;
  1358→
  1359→        continue;
  1360→
  1361→      }
  1362→
  1363→      // Simplified fire check: if enough ticks have passed and player has some knowledge
  1364→
  1365→      if (session >= beat.sessionRange[0] && ws.knowledgeFacts.length > 0) {
  1366→
  1367→        beat.fired = true;
  1368→
  1369→        beat.firedAtTick = currentTick;
  1370→
  1371→        spine.beatsCompleted.push(beat.beatId);
  1372→
  1373→        fired.push(beat.beatId);
  1374→
  1375→      }
  1376→
  1377→    }
  1378→
  1379→    // Check for drift mode
  1380→
  1381→    const missedInAct = spine.beatsMissed.filter(b => {
  1382→
  1383→      return act.beats.some(ab => ab.beatId === b);
  1384→
  1385→    }).length;
  1386→
  1387→    if (missedInAct >= spine.maxMissedPerAct) {
  1388→
  1389→      spine.driftMode = true;
  1390→
  1391→    }
  1392→
  1393→    // Advance act

  1395→    if (act.beats.every(b => b.fired) && spine.currentAct < 3) {
  1396→
  1397→      spine.currentAct = (spine.currentAct + 1) as 1 | 2 | 3;
  1398→
  1399→    }
  1400→
  1401→  }
  1402→
  1403→    return fired;
  1404→
  1405→}
