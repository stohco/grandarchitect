   527→function generateProceduralNpc(
   528→  rng: () => bigint,
   529→  entityId: EntityId,
   530→  householdIdx: number,
   531→  isWangHousehold: boolean,
   532→  wealthTier: WealthTier,
   533→  positionInHousehold: number,
   534→): GeneratedNpc {
   535→  const isWang = isWangHousehold && rngFloat(rng, 0, 1) < 0.85;
   536→  const surnamePool = isWang
   537→    ? [['Wang','Wang'] as const]
   538→    : NON_WANG_SURNAMES.map(s => [s, s] as const);
   539→  const [surname, surnameHanzi] = rngChoice(rng, surnamePool);
   540→
   541→  const sex: Sex = rngChoice(rng, ['male','female']);
   542→  const age = positionInHousehold === 0
   543→    ? rngInt(rng, 30, 65)  // household head
   544→    : rngInt(rng, 0, 75);
   545→
   546→  const givenPool = isWang
   547→    ? (sex === 'male' ? WANG_GIVEN_MALE : WANG_GIVEN_FEMALE)
   548→    : (sex === 'male' ? NON_WANG_GIVEN_MALE : NON_WANG_GIVEN_FEMALE);
   549→  const givenName = rngChoice(rng, givenPool);
   550→  const givenNameHanzi = givenName;
   551→
   552→  const role = positionInHousehold === 0
   553→    ? (isWang ? rngChoice(rng, ['yeoman_farmer','tenant_farmer'] as NpcRole[]) : rngChoice(rng, PROCEDURAL_ROLES))
   554→    : (age < 15
   555→      ? (sex === 'male' ? 'son' as NpcRole : 'daughter' as NpcRole)
   556→      : age >= 60
   557→        ? (sex === 'male' ? 'elder_dependent' as NpcRole : 'elder_dependent' as NpcRole)
   558→        : rngChoice(rng, ['wife','son','daughter','weaver','day_laborer','apprentice'] as NpcRole[]));
   559→
   560→  const name = `${surname} ${givenName}`;
   561→  const nameHanzi = `${surnameHanzi}${givenNameHanzi}`;
   562→
   563→  const traits = generateTraits(rng, role);
   564→  const skills = PROCEDURAL_SKILLS[role] ?? ['farming'];
   565→
   566→  return {
   567→    entityId,
   568→    name,
   569→    nameHanzi,
   570→    surname,
   571→    surnameHanzi,
   572→    givenName,
   573→    givenNameHanzi,
   574→    age,
   575→    sex,
   576→    lifeStage: lifeStageFromAge(age),
   577→    role,
   578→    householdIndex: householdIdx,
   579→    isWang,
   580→    isNamed: false,
   581→    wealthTier,
   582→    qiState: 'none',
   583→    traits,
   584→    skills: [...skills],
   585→    kinship: [],
   586→    relationships: [],
   587→    personalitySummary: `Procedural ${role} of household ${householdIdx}.`,
   588→    tensionTies: [],
   589→    metadata: {},
   590→  };
   591→}

   714→  rel('Wang Zongde','Wang Sanniang','rival_tension',-0.3,-0.2);    // "bare stick" tension
   715→}

   717→// ============================================================================
   718→// Main generation function
   719→// ============================================================================
   720→
   721→export function generateNpcRoster(params: NpcGenParams): NpcRoster {
   722→  const seed = params.seed;
   723→  const rng = createRng(hashSeedToU64(seed + ':npc'));
   724→  const totalPop = params.totalPopulation ?? 180;
   725→  const namedOnly = params.namedOnly ?? false;
   726→  const householdCount = params.householdCount ?? 31;
   727→
   728→  const npcs: GeneratedNpc[] = [];
   729→  let entityIdCounter = 0n;
   730→  function nextEntityId(): EntityId { return entityIdCounter++; }
   731→
   732→  // ---- 1. Generate named NPCs from doc 34 ----
   733→  for (const tmpl of NAMED_NPCS) {
   734→    npcs.push({
   735→      entityId: nextEntityId(),
   736→      name: tmpl.name,
   737→      nameHanzi: tmpl.nameHanzi,
   738→      surname: tmpl.surname,
   739→      surnameHanzi: tmpl.surnameHanzi,
   740→      givenName: tmpl.givenName,
   741→      givenNameHanzi: tmpl.givenNameHanzi,
   742→      age: tmpl.age,
   743→      sex: tmpl.sex,
   744→      lifeStage: lifeStageFromAge(tmpl.age),
   745→      role: tmpl.role,
   746→      householdIndex: tmpl.householdIndex,
   747→      isWang: tmpl.isWang,
   748→      isNamed: true,
   749→      wealthTier: tmpl.wealthTier,
   750→      qiState: tmpl.qiState,
   751→      traits: generateTraits(rng, tmpl.role),
   752→      skills: [...tmpl.skills],
   753→      kinship: [],
   754→      relationships: [],
   755→      personalitySummary: tmpl.personalitySummary,
   756→      tensionTies: [...tmpl.tensionTies],
   757→      metadata: {},
   758→    });
   759→  }
   760→
   761→  // Wire kinship and relationships among named NPCs
   762→  wireNamedKinship(npcs);
   763→  wireRelationships(npcs);
   764→
   765→  if (namedOnly) {
   766→    return {
   767→      seed,
   768→      tick: 0,
   769→      totalGenerated: npcs.length,
   770→      namedCount: npcs.length,
   771→      proceduralCount: 0,
   772→      npcs,
   773→    };
   774→  }
   775→
   776→  // ---- 2. Generate procedural NPCs for remaining households ----
   777→  // The first 15 households have named heads; the remaining 16 are fully procedural.
   778→  // Each household has 3-8 members (average ~5.8 for 180 people / 31 households).
   779→  const namedHouseholdIndices = new Set(NAMED_NPCS.map(n => n.householdIndex));
   780→  const proceduralPopTarget = totalPop - npcs.length;
   781→  let proceduralGenerated = 0;
   782→
   783→  // Wealth distribution for unnamed households (mostly poor/tenant)
   784→  const unnamedWealthTiers: WealthTier[] = ['poor','poor','poor','comfortable','destitute'];
   785→
   786→  for (let hh = 0; hh < householdCount && proceduralGenerated < proceduralPopTarget; hh++) {
   787→    const isNamedHH = namedHouseholdIndices.has(hh);
   788→    const wt = isNamedHH
   789→      ? (NAMED_NPCS.find(n => n.householdIndex === hh)?.wealthTier ?? 'poor')
   790→      : rngChoice(rng, unnamedWealthTiers);
   791→    const isWangHH = isNamedHH
   792→      ? (NAMED_NPCS.some(n => n.householdIndex === hh && n.isWang))
   793→      : (hh < 22); // 22 of 31 are Wang per doc 04
   794→
   795→    // Count existing NPCs in this household
   796→    const existingInHH = npcs.filter(n => n.householdIndex === hh).length;
   797→    const targetMembers = isNamedHH
   798→      ? rngInt(rng, 1, 4)  // add a few more members to named households
   799→      : rngInt(rng, 3, 8);  // full household for unnamed ones
   800→
   801→    for (let i = 0; i < targetMembers && proceduralGenerated < proceduralPopTarget; i++) {
   802→      const npc = generateProceduralNpc(
   803→        rng, nextEntityId(), hh, isWangHH, wt, existingInHH + i,
   804→      );
   805→      npcs.push(npc);
   806→      proceduralGenerated++;
   807→    }
   808→  }
   809→
   810→  return {
   811→    seed,
   812→    tick: 0,
   813→    totalGenerated: npcs.length,
   814→    namedCount: NAMED_NPCS.length,
   815→    proceduralCount: proceduralGenerated,
   816→    npcs,
   817→  };
   818→}

   821→// Query API
   822→// ============================================================================
   823→
   824→export interface NpcGenApi {
   825→  generate(params: NpcGenParams): NpcRoster;
   826→  getLast(): NpcRoster | null;
   827→  stats(): NpcGenStats;
   828→  queryNpc(entityId: EntityId): GeneratedNpc | undefined;
   829→  queryByHousehold(householdIndex: number): GeneratedNpc[];
   830→  queryByRole(role: NpcRole): GeneratedNpc[];
   831→  queryNamed(): GeneratedNpc[];
   832→  queryByTension(tensionId: string): GeneratedNpc[];
   833→  getKinshipNetwork(entityId: EntityId): { npc: GeneratedNpc; kin: GeneratedNpc[]; rels: GeneratedNpc[] };
   834→}
   835→
   836→export interface NpcGenStats {
   837→  generationsRun: number;
   838→  lastSeed: string | null;
   839→  lastTotalNpcs: number;
   840→  lastNamedCount: number;
   841→}
   842→
   843→export function createNpcGenApi(): NpcGenApi {
   844→  let last: NpcRoster | null = null;
   845→  let generationsRun = 0;
   846→
   847→  return {
   848→    generate(params: NpcGenParams): NpcRoster {
   849→      generationsRun++;
   850→      last = generateNpcRoster(params);
   851→      return last;
   852→    },
   853→    getLast(): NpcRoster | null { return last; },
   854→    stats(): NpcGenStats {
   855→      return {
   856→        generationsRun,
   857→        lastSeed: last?.seed ?? null,
   858→        lastTotalNpcs: last?.totalGenerated ?? 0,
   859→        lastNamedCount: last?.namedCount ?? 0,
   860→      };
   861→    },
   862→    queryNpc(entityId: EntityId): GeneratedNpc | undefined {
   863→      return last?.npcs.find(n => n.entityId === entityId);
   864→    },
   865→    queryByHousehold(householdIndex: number): GeneratedNpc[] {
   866→      return last?.npcs.filter(n => n.householdIndex === householdIndex) ?? [];
   867→    },
   868→    queryByRole(role: NpcRole): GeneratedNpc[] {
   869→      return last?.npcs.filter(n => n.role === role) ?? [];
   870→    },
   871→    queryNamed(): GeneratedNpc[] {
   872→      return last?.npcs.filter(n => n.isNamed) ?? [];
   873→    },
   874→    queryByTension(tensionId: string): GeneratedNpc[] {
   875→      return last?.npcs.filter(n => n.tensionTies.includes(tensionId)) ?? [];
   876→    },
   877→    getKinshipNetwork(entityId: EntityId) {
   878→      if (!last) return { npc: undefined as any, kin: [], rels: [] };
   879→      const npc = last.npcs.find(n => n.entityId === entityId);
   880→      if (!npc) return { npc: undefined as any, kin: [], rels: [] };
   881→      const kinIds = new Set(npc.kinship.map(k => k.targetNpcId));
   882→      const relIds = new Set(npc.relationships.map(r => r.targetNpcId));
   883→      return {
   884→        npc,
   885→        kin: last.npcs.filter(n => kinIds.has(n.entityId)),
   886→        rels: last.npcs.filter(n => relIds.has(n.entityId)),
   887→      };
   888→    },
   889→  };
   890→}

   897→  let api: NpcGenApi | null = null;
