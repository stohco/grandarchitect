    81→
    82→function assert(condition: boolean, label: string) {
    83→  if (condition) {
    84→    passed++;
    85→    console.log(`  \u2705 ${label}`);
    86→  } else {
    87→    failed++;
    88→    console.error(`  \u274c ${label}`);
    89→  }
    90→}
    91→
    92→function assertEq<T>(actual: T, expected: T, label: string) {
    93→  if (JSON.stringify(actual) === JSON.stringify(expected)) {
    94→    passed++;
    95→    console.log(`  \u2705 ${label}`);
    96→  } else {
    97→    failed++;
    98→    console.error(`  \u274c ${label} — expected ${JSON.stringify(expected)}, got ${JSON.stringify(actual)}`);
    99→  }
   100→}
   101→
   102→function approxEq(a: number, b: number, eps: number, label: string) {
   103→  const diff = a - b;
   104→  const abs = diff >= 0 ? diff : -diff;
   105→  if (abs <= eps) {
   106→    passed++;
   107→    console.log(`  \u2705 ${label}`);
   108→  } else {
   109→    failed++;
   110→    console.error(`  \u274c ${label} — expected ~${b}, got ${a}, diff=${abs}`);
   111→  }
   112→}
