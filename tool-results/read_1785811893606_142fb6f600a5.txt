   693→   693→export function evaluateCondition(cond: DialogueCondition, ws: WorldStateSnapshot): boolean {
   694→   694→
   695→   695→  switch (cond.type) {
   696→   696→
   697→   697→    case 'perception_state':
   698→   698→
   699→   699→      return ws.perceptionState === (cond.state as PerceptionState);
   700→   700→
   701→   701→    case 'knowledge_fact':
   702→   702→
   703→   703→      return cond.factId ? ws.knowledgeFacts.includes(cond.factId) : false;
   704→   704→
   705→   705→    case 'relationship':
   706→   706→
   707→   707→      if (!cond.target) return false;
   708→   708→
   709→   709→      const rel = ws.relationships[keyOf(cond.target)];
   710→   710→
   711→   711→      return rel !== undefined && rel >= (cond.min ?? 0);
   712→   712→
   713→   713→    case 'realm': {
   714→   714→
   715→   715→      const realmOrder: Realm[] = ['mortal', 'qi_induction', 'qi_condensation', 'foundation_establishment', 'core_formation', 'nascent_soul', 'spirit_severance', 'void_amalgamation', 'tribulation_crossing', 'mahayana'];
   716→   716→
   717→   717→      const playerIdx = realmOrder.indexOf(ws.playerRealm);
   718→   718→
   719→   719→      const minIdx = realmOrder.indexOf((cond.state ?? 'mortal') as Realm);
   720→   720→
   721→   721→      return playerIdx >= minIdx;
   722→   722→
   723→   723→    }
   724→   724→
   725→   725→    case 'quest_state':
   726→   726→
   727→   727→      return true; // Simplified — quests are managed by the API
   728→   728→
   729→   729→    case 'flag':
   730→   730→
   731→   731→      return cond.flag ? (ws.flags[cond.flag] ?? false) === (cond.value ?? true) : false;
   732→   732→
   733→   733→    case 'inventory':
   734→   734→
   735→   735→      return cond.itemId ? (ws.inventory[cond.itemId] ?? 0) >= (cond.min ?? 1) : false;
   736→   736→
   737→   737→    default:
   738→   738→
   739→   739→      return true;
   740→   740→
   741→   741→  }

   805→   805→export function computeSmoothedApproval(approval: number, history: SharedExperience[], windowDays: number): number {
   806→   806→
   807→   807→  if (history.length === 0) return approval;
   808→   808→
   809→   809→  const recentDelta = history
   810→   810→
   811→   811→    .slice(-10) // simplified: last 10 experiences
   812→   812→
   813→   813→    .reduce((sum, e) => sum + e.approvalDelta, 0);
   814→   814→
   815→   815→  const weight = Math.min(1, history.length / 10);
   816→   816→
   817→   817→  return approval * (1 - weight * 0.3) + (approval + recentDelta / 10) * (weight * 0.3);
   818→   818→
   819→   819→}
